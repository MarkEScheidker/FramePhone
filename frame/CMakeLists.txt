cmake_minimum_required(VERSION 3.20)
project(framephone_frame LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)

option(FRAMEPHONE_BUILD_OPENVR_OVERLAY "Build the existing OpenVR demo overlay" OFF)
option(FRAMEPHONE_ENABLE_APRILTAG "Build real AprilTag 3 tag36h11 detection when installed" ON)
option(FRAMEPHONE_ENABLE_SDL2 "Enable the desktop diagnostic preview when SDL2 is installed" ON)

add_library(framephone_camera STATIC src/camera_source.cpp src/frametop_ring.cpp)
target_include_directories(framephone_camera PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_options(framephone_camera PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)

add_executable(framephone-frame src/tracker_main.cpp)
target_link_libraries(framephone-frame PRIVATE framephone_camera)
target_compile_options(framephone-frame PRIVATE -Wall -Wextra -Wpedantic)

add_library(framephone_diagnostic STATIC src/diagnostic.cpp)
target_link_libraries(framephone_diagnostic PUBLIC framephone_camera)
target_compile_options(framephone_diagnostic PRIVATE -Wall -Wextra -Wpedantic)
add_executable(framephone-diagnostic src/diagnostic_main.cpp)
target_link_libraries(framephone-diagnostic PRIVATE framephone_diagnostic)
target_compile_options(framephone-diagnostic PRIVATE -Wall -Wextra -Wpedantic)

option(FRAMEPHONE_ENABLE_POSE "Enable calibrated planar pose with OpenCV and nlohmann JSON" ON)
if(FRAMEPHONE_ENABLE_POSE)
    set(POSE_SDK_ROOT "" CACHE PATH "Optional local OpenCV/JSON install prefix")
    find_path(POSE_OPENCV_INCLUDE opencv2/calib3d.hpp HINTS "${POSE_SDK_ROOT}" PATH_SUFFIXES include/opencv4)
    find_path(POSE_JSON_INCLUDE nlohmann/json.hpp HINTS "${POSE_SDK_ROOT}" PATH_SUFFIXES include)
    find_library(POSE_OPENCV_CORE opencv_core HINTS "${POSE_SDK_ROOT}" PATH_SUFFIXES lib "lib/${CMAKE_LIBRARY_ARCHITECTURE}")
    find_library(POSE_OPENCV_CALIB opencv_calib3d HINTS "${POSE_SDK_ROOT}" PATH_SUFFIXES lib "lib/${CMAKE_LIBRARY_ARCHITECTURE}")
    if(POSE_OPENCV_INCLUDE AND POSE_JSON_INCLUDE AND POSE_OPENCV_CORE AND POSE_OPENCV_CALIB)
        add_library(framephone_pose STATIC src/pose.cpp src/frametop_calibration.cpp src/stereo.cpp)
        target_link_libraries(framephone_pose PUBLIC framephone_camera PRIVATE "${POSE_OPENCV_CALIB}" "${POSE_OPENCV_CORE}")
        target_include_directories(framephone_pose SYSTEM PRIVATE "${POSE_OPENCV_INCLUDE}" "${POSE_JSON_INCLUDE}")
        target_compile_options(framephone_pose PRIVATE -Wall -Wextra -Wpedantic)
        target_link_libraries(framephone-diagnostic PRIVATE framephone_pose)
        target_compile_definitions(framephone-diagnostic PRIVATE FRAMEPHONE_HAS_POSE=1)
        message(STATUS "OpenCV/JSON found: camera-relative planar pose enabled")
    else()
        message(STATUS "OpenCV/JSON not found: pose options report unavailable")
    endif()
endif()

if(FRAMEPHONE_ENABLE_SDL2)
    set(SDL2_ROOT "" CACHE PATH "SDL2 install prefix")
    find_path(SDL2_INCLUDE_DIR SDL.h HINTS "${SDL2_ROOT}" PATH_SUFFIXES include/SDL2)
    find_library(SDL2_LIBRARY SDL2 HINTS "${SDL2_ROOT}" PATH_SUFFIXES lib "lib/${CMAKE_LIBRARY_ARCHITECTURE}")
    if(SDL2_INCLUDE_DIR AND SDL2_LIBRARY)
        target_sources(framephone-diagnostic PRIVATE src/diagnostic_view.cpp)
        target_include_directories(framephone-diagnostic PRIVATE "${SDL2_INCLUDE_DIR}")
        # Debian/Ubuntu SDL_config.h delegates to a multiarch header. This path
        # is normally provided by the compiler, but also supports a local SDK.
        if(EXISTS "${SDL2_ROOT}/include/${CMAKE_LIBRARY_ARCHITECTURE}/SDL2/_real_SDL_config.h")
            target_include_directories(framephone-diagnostic PRIVATE "${SDL2_ROOT}/include/${CMAKE_LIBRARY_ARCHITECTURE}")
        endif()
        target_link_libraries(framephone-diagnostic PRIVATE "${SDL2_LIBRARY}")
        target_compile_definitions(framephone-diagnostic PRIVATE FRAMEPHONE_HAS_SDL2=1 SDL_MAIN_HANDLED=1)
        message(STATUS "SDL2 found: native diagnostic preview enabled")
    else()
        message(STATUS "SDL2 not found: diagnostic builds with --headless support")
    endif()
endif()

if(FRAMEPHONE_ENABLE_APRILTAG)
    set(APRILTAG_ROOT "" CACHE PATH "AprilTag 3 source or install prefix")
    find_path(APRILTAG_INCLUDE_DIR NAMES apriltag/apriltag.h apriltag.h HINTS "${APRILTAG_ROOT}" PATH_SUFFIXES include)
    find_library(APRILTAG_LIBRARY NAMES apriltag HINTS "${APRILTAG_ROOT}" PATH_SUFFIXES lib lib64)
    if(APRILTAG_INCLUDE_DIR AND APRILTAG_LIBRARY)
        add_library(framephone_apriltag STATIC src/apriltag_detector.cpp)
        target_include_directories(framephone_apriltag PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" PRIVATE "${APRILTAG_INCLUDE_DIR}")
        if(EXISTS "${APRILTAG_INCLUDE_DIR}/apriltag/apriltag.h")
            target_compile_definitions(framephone_apriltag PRIVATE FRAMEPHONE_APRILTAG_NESTED_HEADERS=1)
        endif()
        target_link_libraries(framephone_apriltag PUBLIC framephone_camera PRIVATE "${APRILTAG_LIBRARY}")
        target_compile_definitions(framephone-frame PRIVATE FRAMEPHONE_HAS_APRILTAG=1)
        target_link_libraries(framephone-frame PRIVATE framephone_apriltag)
        target_compile_definitions(framephone-diagnostic PRIVATE FRAMEPHONE_HAS_APRILTAG=1)
        target_link_libraries(framephone-diagnostic PRIVATE framephone_apriltag)
        message(STATUS "AprilTag 3 found: tag36h11 detector enabled")
        if(BUILD_TESTING)
            add_executable(framephone-apriltag-tests tests/test_apriltag_detector.cpp)
            target_link_libraries(framephone-apriltag-tests PRIVATE framephone_apriltag)
            target_compile_options(framephone-apriltag-tests PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    else()
        message(STATUS "AprilTag 3 not found: replay/synthetic sources build; detection reports unavailable")
    endif()
endif()

option(FRAMEPHONE_REQUIRE_TRACKER "Fail configuration unless real stereo tracker dependencies exist" OFF)
if(TARGET framephone_pose AND TARGET framephone_apriltag)
    add_executable(framephone-stereo src/stereo_main.cpp)
    target_link_libraries(framephone-stereo PRIVATE framephone_pose framephone_apriltag framephone_diagnostic)
    target_compile_options(framephone-stereo PRIVATE -Wall -Wextra -Wpedantic)
    install(TARGETS framephone-stereo framephone-diagnostic RUNTIME DESTINATION bin)
    install(FILES THIRD_PARTY_NOTICES.md DESTINATION share/framephone)
    install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE" DESTINATION share/framephone)
    install(FILES third_party/frametop/LICENSE third_party/frametop/README.md
        DESTINATION share/framephone/third_party/frametop)
    install(FILES docs/HEADSET_TEST.md docs/STEREO.md DESTINATION share/framephone/docs)
elseif(FRAMEPHONE_REQUIRE_TRACKER)
    message(FATAL_ERROR "Tracker requires AprilTag 3, OpenCV core/calib3d and nlohmann JSON development files. No detector-free fallback is allowed for this build.")
endif()

if(FRAMEPHONE_BUILD_OPENVR_OVERLAY)
    set(OPENVR_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/include" CACHE PATH "OpenVR SDK headers")
    set(OPENVR_LIBRARY_DIR "/opt/steamvr/bin/linuxarm64" CACHE PATH "SteamVR OpenVR library directory")
    find_path(OPENVR_HEADER_DIR openvr.h HINTS "${OPENVR_INCLUDE_DIR}" REQUIRED)
    find_library(OPENVR_LIBRARY NAMES openvr_api HINTS "${OPENVR_LIBRARY_DIR}" REQUIRED)
    add_executable(framephone-openvr-demo src/main.cpp)
    target_include_directories(framephone-openvr-demo PRIVATE "${OPENVR_HEADER_DIR}")
    target_link_libraries(framephone-openvr-demo PRIVATE "${OPENVR_LIBRARY}" ${CMAKE_DL_LIBS})
    target_compile_options(framephone-openvr-demo PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(framephone-openvr-demo PROPERTIES
        BUILD_RPATH "${OPENVR_LIBRARY_DIR}"
        INSTALL_RPATH "${OPENVR_LIBRARY_DIR}")
endif()

        target_compile_definitions(framephone-pose-tests PRIVATE FRAMEPHONE_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
        target_compile_options(framephone-pose-tests PRIVATE -Wall -Wextra -Wpedantic)
        add_test(NAME framephone-calibrated-pose COMMAND framephone-pose-tests)
        add_executable(framephone-stereo-tests tests/test_stereo.cpp)
        target_link_libraries(framephone-stereo-tests PRIVATE framephone_pose framephone_apriltag "${POSE_OPENCV_CORE}" "${POSE_OPENCV_CALIB}" "${POSE_OPENCV_IMGPROC}")
        target_include_directories(framephone-stereo-tests SYSTEM PRIVATE "${POSE_OPENCV_INCLUDE}" "${POSE_JSON_INCLUDE}")

        target_compile_options(framephone-stereo-tests PRIVATE -Wall -Wextra -Wpedantic)
        add_test(NAME framephone-stereo-geometry COMMAND framephone-stereo-tests)

    add_executable(framephone-diagnostic-tests tests/test_diagnostic.cpp)
    target_link_libraries(framephone-diagnostic-tests PRIVATE framephone_diagnostic)
    target_compile_options(framephone-diagnostic-tests PRIVATE -Wall -Wextra -Wpedantic)

            "$<TARGET_FILE:framephone-frame>")

            add_test(NAME framephone-stereo-cli COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_stereo_cli.py" "$<TARGET_FILE:framephone-stereo>"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
        endif()

    add_executable(framephone-camera-tests tests/test_camera_source.cpp)
    target_link_libraries(framephone-camera-tests PRIVATE framephone_camera)
    target_compile_options(framephone-camera-tests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME framephone-camera-sources COMMAND framephone-camera-tests)
    if(TARGET framephone-apriltag-tests)
        add_test(NAME framephone-apriltag-synthetic-detections COMMAND framephone-apriltag-tests)
