cmake_minimum_required(VERSION 3.20)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(framephone_camera_module LANGUAGES CXX)
    include(CTest)
    set(camera_standalone ON)
else()
    set(camera_standalone OFF)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The Frame camera module requires Linux")
endif()

option(FRAMEPHONE_CAMERA_BUILD_EXAMPLE "Build the camera reader example" ${camera_standalone})
option(FRAMEPHONE_CAMERA_BUILD_BROKER "Build the Steam Frame camera broker" OFF)
option(FRAMEPHONE_CAMERA_STATIC_BROKER "Link the broker statically (requires static libc)" OFF)

add_library(framephone_camera STATIC src/camera_source.cpp src/frametop_ring.cpp)
add_library(FramePhone::Camera ALIAS framephone_camera)
target_compile_features(framephone_camera PUBLIC cxx_std_20)
set_target_properties(framephone_camera PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(framephone_camera PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
    "$<INSTALL_INTERFACE:include>")
target_compile_options(framephone_camera PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)

if(FRAMEPHONE_CAMERA_BUILD_EXAMPLE)
    add_executable(framephone-camera-read examples/read_frames.cpp)
    target_link_libraries(framephone-camera-read PRIVATE FramePhone::Camera)
endif()

if(FRAMEPHONE_CAMERA_BUILD_BROKER)
    enable_language(C)
    add_executable(framephone-camd broker/camd.c broker/tp.c broker/xrcams.c)
    set_target_properties(framephone-camd PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS ON)
    target_compile_options(framephone-camd PRIVATE -O2 -Wall -Wextra)
    target_link_libraries(framephone-camd PRIVATE m)
    if(FRAMEPHONE_CAMERA_STATIC_BROKER)
        target_link_options(framephone-camd PRIVATE -static)
    endif()
    install(TARGETS framephone-camd RUNTIME DESTINATION bin)
endif()

if(BUILD_TESTING)
    foreach(test camera_source frametop_ring)
        add_executable(framephone-${test}-tests tests/test_${test}.cpp)
        target_link_libraries(framephone-${test}-tests PRIVATE FramePhone::Camera)
        target_compile_options(framephone-${test}-tests PRIVATE -Wall -Wextra -Wpedantic)
        add_test(NAME framephone-${test} COMMAND framephone-${test}-tests)
    endforeach()
    if(FRAMEPHONE_CAMERA_BUILD_EXAMPLE)
        add_test(NAME framephone-camera-example-help COMMAND framephone-camera-read --help)
    endif()
    if(FRAMEPHONE_CAMERA_BUILD_BROKER)
        add_test(NAME framephone-broker-help COMMAND framephone-camd --help)
    endif()
endif()

if(camera_standalone)
    install(TARGETS framephone_camera ARCHIVE DESTINATION lib)
    install(DIRECTORY include/ DESTINATION include)
endif()
install(FILES LICENSE README.md RECORDINGS.md THIRD_PARTY_NOTICES.md
    DESTINATION share/framephone/modules/camera)
install(DIRECTORY broker/ DESTINATION share/framephone/modules/camera/broker)
